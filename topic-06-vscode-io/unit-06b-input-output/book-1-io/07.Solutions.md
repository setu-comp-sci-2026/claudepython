## Exercise 1 — BMI Calculator

```python
def get_float(prompt, minimum=0):
    while True:
        try:
            value = float(input(prompt))
            if value <= minimum:
                print(f"Please enter a value greater than {minimum}")
                continue
            return value
        except ValueError:
            print("Please enter a number")

def bmi_category(bmi):
    if bmi < 18.5: return "Underweight"
    if bmi < 25.0: return "Normal weight"
    if bmi < 30.0: return "Overweight"
    return "Obese"

name   = input("Name: ").strip()
weight = get_float("Weight (kg): ")
height = get_float("Height (m): ")

bmi = weight / (height ** 2)
category = bmi_category(bmi)

print(f"\nBMI Report for {name}")
print(f"Weight: {weight:.1f} kg")
print(f"Height: {height:.2f} m")
print(f"BMI:    {bmi:.1f}  ({category})")
```

## Exercise 2 — Currency Converter

```python
rates = {"EUR": 1.00, "USD": 1.08, "GBP": 0.86}
symbols = {"EUR": "€", "USD": "$", "GBP": "£"}

def convert(amount, from_currency, to_currency):
    in_eur = amount / rates[from_currency]
    return in_eur * rates[to_currency]

while True:
    print(f"\nCurrencies: {', '.join(rates.keys())}")
    from_cur = input("From currency (or 'quit'): ").upper()
    if from_cur == "QUIT":
        break
    if from_cur not in rates:
        print("Unknown currency")
        continue
    to_cur = input("To currency: ").upper()
    if to_cur not in rates:
        print("Unknown currency")
        continue
    try:
        amount = float(input(f"Amount ({symbols[from_cur]}): "))
        result = convert(amount, from_cur, to_cur)
        print(f"{symbols[from_cur]}{amount:.2f} = {symbols[to_cur]}{result:.2f}")
    except ValueError:
        print("Invalid amount")
```

## Exercise 3 — Receipt Printer

```python
items = []
print("Enter items (name price). Type 'done' to finish:")

while True:
    entry = input("> ").strip()
    if entry.lower() == "done":
        break
    parts = entry.rsplit(" ", 1)

            name, price = parts[0], float(parts[1])
            items.append((name, price))
        except ValueError:
            print("Format: item_name price")

        print("Format: item_name price")

if not items:
    print("No items entered")
else:
    subtotal = sum(price for _, price in items)
    discount = subtotal * 0.10 if subtotal > 50 else 0

    width = 30
```

## Exercise 4 — Grade Report

```python
def letter_grade(score):
    if score >= 90: return "A"
    if score >= 80: return "B"
    if score >= 70: return "C"
    if score >= 60: return "D"
    return "F"

students = []
print("Enter student name and score (blank name to finish):")
while True:
    name = input("Name: ").strip()
    if not name:
        break
    try:
        score = float(input("Score: "))
        students.append((name, score))
    except ValueError:
        print("Invalid score")

students.sort(key=lambda s: s[1], reverse=True)

print(f"\n{'Rank':<5} {'Name':<15} {'Score':>7} {'Grade':>6} {'Status':>7}")
print("-" * 43)
for rank, (name, score) in enumerate(students, 1):
    grade = letter_grade(score)
    status = "Pass" if score >= 60 else "Fail"
    print(f"{rank:<5} {name:<15} {score:>7.1f} {grade:>6} {status:>7}")
```
