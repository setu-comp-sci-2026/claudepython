# Writing Files

## Step 1 — Write a Simple File

```python
lines = [
    "Python Programming Lab 05",
    "=" * 30,
    "Topic: File Handling",
    "Date: 2024",
]

with open("notes.txt", "w") as f:
    for line in lines:
        f.write(line + "\n")

print("File written successfully")

# Verify by reading it back
with open("notes.txt", "r") as f:
    print(f.read())
```

## Step 2 — Write Results to a Report File

```python
students = [
    ("Alice", [88, 92, 78]),
    ("Bob",   [74, 68, 81]),
    ("Carol", [95, 91, 98]),
    ("Dave",  [62, 55, 70]),
]

with open("report.txt", "w") as f:
    f.write("Student Results Report\n")
    f.write("=" * 40 + "\n\n")

    for name, scores in students:
        avg = sum(scores) / len(scores)
        grade = "Pass" if avg >= 60 else "Fail"
        f.write(f"Name:   {name}\n")
        f.write(f"Scores: {scores}\n")
        f.write(f"Average:{avg:.1f}  ({grade})\n")
        f.write("-" * 40 + "\n")

print("Report saved to report.txt")
```

## Step 3 — Append to a Log File

```python
import datetime

def log(message):
    timestamp = datetime.datetime.now().strftime("%H:%M:%S")
    with open("activity.log", "a") as f:
        f.write(f"[{timestamp}] {message}\n")

log("Program started")
log("User logged in")
log("File processed: data.csv")
log("Program ended")

# Read and display the log
print("Activity Log:")
with open("activity.log", "r") as f:
    print(f.read())
```

## Step 4 — Round-Trip: Read, Process, Write

```python
# Read scores, apply a bonus, write updated scores

# Read
students = []
with open("scores.txt", "r") as f:
    for line in f:
        parts = line.strip().split()
        if len(parts) == 2:
            students.append([parts[0], int(parts[1])])

# Process — apply 5% bonus
for student in students:
    student[1] = min(100, round(student[1] * 1.05))

# Write updated scores
with open("scores_updated.txt", "w") as f:
    for name, score in students:
        f.write(f"{name} {score}\n")

print("Updated scores written")
for name, score in students:
    print(f"  {name}: {score}")
```
