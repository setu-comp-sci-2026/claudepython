# Reading Files

## Setup — Create a Sample File

First, create a file called `scores.txt` with this content (or create it programmatically):

```python
# Run this once to create the sample file
with open("scores.txt", "w") as f:
    f.write("Alice 88\n")
    f.write("Bob 74\n")
    f.write("Carol 95\n")
    f.write("Dave 62\n")
    f.write("Eve 81\n")
```

## Step 1 — Read the Entire File

```python
with open("scores.txt", "r") as f:
    content = f.read()

print(content)
print(f"Total characters: {len(content)}")
```

## Step 2 — Read Line by Line

```python
with open("scores.txt", "r") as f:
    for line in f:
        line = line.strip()       # remove trailing \n
        if line:                  # skip blank lines
            parts = line.split()
            name = parts[0]
            score = int(parts[1])
            grade = "Pass" if score >= 60 else "Fail"
            print(f"{name:<10} {score:3}  {grade}")
```

## Step 3 — Read All Lines into a List

```python
with open("scores.txt", "r") as f:
    lines = [line.strip() for line in f if line.strip()]

print(f"Read {len(lines)} lines")
for line in lines:
    print(line)
```

## Step 4 — Process and Analyse

```python
scores = []

with open("scores.txt", "r") as f:
    for line in f:
        parts = line.strip().split()
        if len(parts) == 2:
            scores.append(int(parts[1]))

if scores:
    print(f"Count:   {len(scores)}")
    print(f"Average: {sum(scores)/len(scores):.1f}")
    print(f"Highest: {max(scores)}")
    print(f"Lowest:  {min(scores)}")
```

## Step 5 — Handle Missing Files Gracefully

```python
def read_scores(filename):
    try:
        scores = []
        with open(filename, "r") as f:
            for line in f:
                parts = line.strip().split()
                if len(parts) == 2:
                    scores.append(int(parts[1]))
        return scores
    except FileNotFoundError:
        print(f"File '{filename}' not found")
        return []
    except ValueError:
        print("File contains invalid data")
        return []

data = read_scores("scores.txt")
print(f"Loaded {len(data)} scores")
```
