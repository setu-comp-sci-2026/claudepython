## Exercise 1 — Animal Hierarchy

```python
class Animal:
    def __init__(self, name: str):
        self.__name = name
    def get_name(self) -> str:
        return self.__name
    def speak(self) -> str:
        return "..."
    def __str__(self) -> str:
        return f"{type(self).__name__}: {self.__name}"

class Dog(Animal):
    def speak(self) -> str: return "Woof!"

class Cat(Animal):
    def speak(self) -> str: return "Meow!"

class Parrot(Animal):
    def __init__(self, name: str, phrase: str):
        super().__init__(name)
        self.__phrase = phrase
    def speak(self) -> str: return self.__phrase

animals = [Dog("Rex"), Cat("Whiskers"), Parrot("Polly", "Hello!")]
for a in animals:
    print(f"{a.get_name()} says: {a.speak()}")
```

## Exercise 4 — Shape Hierarchy

```python
import math

class Shape:
    def area(self) -> float: return 0.0

class Circle(Shape):
    def __init__(self, r): self.__r = r
    def area(self): return math.pi * self.__r ** 2

class Rectangle(Shape):
    def __init__(self, w, h): self.__w = w; self.__h = h
    def area(self): return self.__w * self.__h

class Triangle(Shape):
    def __init__(self, b, h): self.__b = b; self.__h = h
    def area(self): return 0.5 * self.__b * self.__h

shapes = [Circle(5), Rectangle(4, 6), Triangle(3, 8)]
total = sum(s.area() for s in shapes)
print(f"Total area: {total:.2f}")
```

## Exercise 3 — Employee Hierarchy

```python
class Employee:
    def __init__(self, name: str, salary: float):
        self.__name = name
        self.__salary = salary
    def get_name(self): return self.__name
    def annual_pay(self): return self.__salary

class Manager(Employee):
    def __init__(self, name, salary, department, team_size):
        super().__init__(name, salary)
        self.__department = department
        self.__team_size = team_size
    def annual_pay(self):
        return super().annual_pay() * 1.1  # 10% bonus

class Contractor(Employee):
    def __init__(self, name, hourly_rate, hours_worked):
        super().__init__(name, 0)
        self.__rate = hourly_rate
        self.__hours = hours_worked
    def annual_pay(self):
        return self.__rate * self.__hours
```
