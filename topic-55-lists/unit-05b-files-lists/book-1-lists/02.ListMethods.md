# List Methods

## Step 1 — Adding Items

```python
todo = []

# append — add to end
todo.append("Buy groceries")
todo.append("Do laundry")
todo.append("Call dentist")
print(todo)

# insert — add at specific position
todo.insert(1, "Walk the dog")   # insert at index 1
print(todo)

# extend — add multiple items
more_tasks = ["Read book", "Exercise"]
todo.extend(more_tasks)
print(todo)
```

## Step 2 — Removing Items

```python
fruits = ["apple", "banana", "cherry", "banana", "date"]

# remove — delete first occurrence of a value
fruits.remove("banana")
print(fruits)   # ['apple', 'cherry', 'banana', 'date']

# pop — remove and return by index
last = fruits.pop()     # removes last item
print(last)
print(fruits)

second = fruits.pop(1)  # removes item at index 1
print(second)
```

## Step 3 — Sorting

```python
scores = [78, 45, 92, 61, 88, 33, 95]

# Sort in place (modifies the list)
scores.sort()
print("Ascending:", scores)

scores.sort(reverse=True)
print("Descending:", scores)

# sorted() — returns new list, original unchanged
names = ["Charlie", "Alice", "Eve", "Bob", "Dave"]
sorted_names = sorted(names)
print("Original:", names)
print("Sorted:", sorted_names)

# Sort by string length
by_length = sorted(names, key=len)
print("By length:", by_length)
```

## Step 4 — Searching

```python
colours = ["red", "green", "blue", "red", "yellow", "red"]

print(colours.count("red"))      # 3
print(colours.count("purple"))   # 0

print(colours.index("green"))    # 1

# Safe index — check before indexing
colour = "purple"
if colour in colours:
    print(f"Found at index {colours.index(colour)}")
else:
    print(f"'{colour}' not in list")
```
