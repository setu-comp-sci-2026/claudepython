## Exercise 1 — Shopping List Manager

```python
shopping = []

while True:
    print(f"\n--- Shopping List ({len(shopping)} items) ---")
    if shopping:
        for i, item in enumerate(shopping, 1):
            print(f"  {i}. {item}")
    print("\n1. Add  2. Remove  3. Show  4. Clear  5. Quit")
    choice = input("Choice: ")

    if choice == "1":
        item = input("Item: ").strip().title()
        if not item:
            print("Item cannot be blank")
        elif item.lower() in [s.lower() for s in shopping]:
            print(f"'{item}' is already on the list")
        else:
            shopping.append(item)
            shopping.sort()
            print(f"Added '{item}'")
    elif choice == "2":
        if not shopping:
            print("List is empty")
        else:
            item = input("Item to remove: ").strip().title()
            if item in shopping:
                shopping.remove(item)
                print(f"Removed '{item}'")
            else:
                print(f"'{item}' not found")
    elif choice == "3":
        if not shopping:
            print("Your list is empty")
    elif choice == "4":
        shopping.clear()
        print("List cleared")
    elif choice == "5":
        break
```

## Exercise 2 — Statistics Calculator

```python
numbers = []
print("Enter numbers (type 'done' to finish):")
while True:
    entry = input("> ")
    if entry.lower() == "done":
        break
    try:
        numbers.append(float(entry))
    except ValueError:
        print("Please enter a number or 'done'")

if numbers:
    numbers_sorted = sorted(numbers)
    n = len(numbers_sorted)
    mean = sum(numbers_sorted) / n
    median = numbers_sorted[n // 2] if n % 2 != 0 else (numbers_sorted[n//2-1] + numbers_sorted[n//2]) / 2
    above_avg = sum(1 for x in numbers if x > mean)

    print(f"\nCount:  {n}")
    print(f"Sum:    {sum(numbers):.2f}")
    print(f"Mean:   {mean:.2f}")
    print(f"Median: {median:.2f}")
    print(f"Min:    {min(numbers):.2f}")
    print(f"Max:    {max(numbers):.2f}")
    print(f"Above average: {above_avg}")
```

## Exercise 3 — Word Frequency from File

```python
def count_words(filename):
    word_count = {}
    with open(filename, "r") as f:
        for line in f:
            for word in line.lower().split():
                clean = "".join(c for c in word if c.isalpha())
                if clean:
                    word_count[clean] = word_count.get(clean, 0) + 1
    return word_count

def write_frequency_report(word_count, output_file):
    sorted_words = sorted(word_count.items(), key=lambda x: (-x[1], x[0]))
    with open(output_file, "w") as f:
        f.write("Word Frequency Report\n")
        f.write("=" * 30 + "\n")
        for word, count in sorted_words:
            f.write(f"{word:<20} {count}\n")

counts = count_words("input.txt")
write_frequency_report(counts, "frequency.txt")
```

## Exercise 4 — File Analysis Tool

```python
def analyse_file(filename):
    try:
        with open(filename, "r") as f:
            lines = f.readlines()
    except FileNotFoundError:
        return None

    non_blank = [l.strip() for l in lines if l.strip()]
    all_words = [w for l in non_blank for w in l.split()]

    return {
        "lines": len(lines),
        "words": len(all_words),
        "chars": sum(len(l.rstrip("\n")) for l in lines),
        "blank_lines": len(lines) - len(non_blank),
        "longest_line": max(lines, key=lambda l: len(l.strip())).strip() if lines else "",
        "avg_words_per_line": len(all_words) / len(non_blank) if non_blank else 0,
    }
```

## Challenge 1 — Student Grade Manager (key parts)

```python
import csv

def load_students(filename):
    students = []
    with open(filename, "r", newline="") as f:
        for row in csv.DictReader(f):
            scores = [int(row["score1"]), int(row["score2"]), int(row["score3"])]
            avg = sum(scores) / 3
            students.append({"name": row["name"], "scores": scores, "avg": avg})
    return students

def letter_grade(avg):
    if avg >= 90: return "A"
    if avg >= 80: return "B"
    if avg >= 70: return "C"
    if avg >= 60: return "D"
    return "F"

students = load_students("students.csv")
for s in students:
    grade = letter_grade(s["avg"])
    status = "Pass" if s["avg"] >= 60 else "FAIL"
    print(f"{s['name']:<10} {s['avg']:5.1f}  {grade}  {status}")
```

## Challenge 3 — Word Search (key parts)

```python
def search_lines(filename, keyword):
    results = []
    with open(filename, "r") as f:
        for i, line in enumerate(f, start=1):
            if keyword.lower() in line.lower():
                results.append((i, line.strip()))
    return results

def replace_in_file(filename, old_word, new_word, output_file):
    with open(filename, "r") as f:
        content = f.read()
    updated = content.replace(old_word, new_word)
    with open(output_file, "w") as f:
        f.write(updated)
    print(f"Replaced '{old_word}' with '{new_word}' in {output_file}")
```
