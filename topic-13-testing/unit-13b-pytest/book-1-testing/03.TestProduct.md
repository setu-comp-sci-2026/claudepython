# Step 3 — Testing the Product Class

Create `tests/test_product.py`:

```python
import pytest
from src.product import Product

@pytest.fixture
def apple():
    return Product("Apple", 1.99, 101)

# --- Construction ---

def test_get_name(apple):
    assert apple.get_name() == "Apple"

def test_get_price(apple):
    assert apple.get_price() == 1.99

def test_get_product_id(apple):
    assert apple.get_product_id() == 101

def test_str(apple):
    assert "Apple" in str(apple)
    assert "1.99" in str(apple)
    assert "101" in str(apple)

# --- Setters ---

def test_set_name(apple):
    apple.set_name("Granny Smith")
    assert apple.get_name() == "Granny Smith"

def test_set_price(apple):
    apple.set_price(2.49)
    assert apple.get_price() == 2.49

# --- Validation ---

def test_empty_name_raises():
    with pytest.raises(ValueError):
        Product("", 1.99, 101)

def test_whitespace_name_raises():
    with pytest.raises(ValueError):
        Product("   ", 1.99, 101)

def test_zero_price_raises():
    with pytest.raises(ValueError):
        Product("Apple", 0.0, 101)

def test_negative_price_raises():
    with pytest.raises(ValueError):
        Product("Apple", -1.0, 101)

# --- Parametrize ---

@pytest.mark.parametrize("name, price, pid", [
    ("Apple",  1.99, 101),
    ("Bread",  2.50, 102),
    ("Milk",   1.20, 103),
    ("Butter", 3.00, 104),
])
def test_valid_products(name, price, pid):
    p = Product(name, price, pid)
    assert p.get_name() == name
    assert p.get_price() == price
    assert p.get_product_id() == pid
```
