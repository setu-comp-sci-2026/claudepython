## Exercise 1 — Word Frequency

```python
# src/text_utils.py
def word_frequency(text: str) -> dict:
    if not text.strip():
        return {}
    freq = {}
    for word in text.lower().split():
        freq[word] = freq.get(word, 0) + 1
    return freq

# tests/test_text_utils.py
from src.text_utils import word_frequency

def test_empty_string():
    assert word_frequency("") == {}

def test_single_word():
    assert word_frequency("hello") == {"hello": 1}

def test_repeated_word():
    result = word_frequency("the cat sat on the mat")
    assert result["the"] == 2
    assert result["cat"] == 1

def test_mixed_case():
    result = word_frequency("Hello hello HELLO")
    assert result["hello"] == 3
```

## TDD Student Solution

```python
# src/student.py
class Student:
    def __init__(self, name: str, email: str,
                 student_id: str):
        self.__name = name
        self.__email = email
        self.__student_id = student_id
        self.__grades = {}

    def get_name(self) -> str:   return self.__name
    def get_email(self) -> str:  return self.__email
    def get_student_id(self) -> str: return self.__student_id

    def add_grade(self, subject: str, mark: float):
        self.__grades[subject] = mark

    def get_grade(self, subject: str) -> float:
        return self.__grades.get(subject, 0.0)

    def average_grade(self) -> float:
        if not self.__grades:
            return 0.0
        return sum(self.__grades.values()) / len(self.__grades)
```
