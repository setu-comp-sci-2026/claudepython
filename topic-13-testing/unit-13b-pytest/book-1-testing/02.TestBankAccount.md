# Step 2 — Testing the BankAccount Class

## 2.1 — Create the BankAccount class

Create `src/bank_account.py`:

```python
class BankAccount:
    def __init__(self, owner: str, balance: float = 0.0):
        if balance < 0:
            raise ValueError("Initial balance cannot be negative")
        self.__owner = owner
        self.__balance = balance

    def get_owner(self) -> str:
        return self.__owner

    def get_balance(self) -> float:
        return self.__balance

    def deposit(self, amount: float):
        if amount <= 0:
            raise ValueError("Deposit amount must be positive")
        self.__balance += amount

    def withdraw(self, amount: float):
        if amount <= 0:
            raise ValueError("Withdrawal amount must be positive")
        if amount > self.__balance:
            raise ValueError(
                f"Insufficient funds: balance is €{self.__balance:.2f}"
            )
        self.__balance -= amount

    def __str__(self) -> str:
        return f"Account({self.__owner}): €{self.__balance:.2f}"
```

## 2.2 — Write tests

Create `tests/test_bank_account.py`:

```python
import pytest
from src.bank_account import BankAccount

@pytest.fixture
def account():
    return BankAccount("Alice", 100.0)

def test_initial_balance(account):
    assert account.get_balance() == 100.0

def test_owner(account):
    assert account.get_owner() == "Alice"

def test_deposit(account):
    account.deposit(50.0)
    assert account.get_balance() == 150.0

def test_multiple_deposits(account):
    account.deposit(10.0)
    account.deposit(20.0)
    assert account.get_balance() == 130.0

def test_withdraw(account):
    account.withdraw(30.0)
    assert account.get_balance() == 70.0

def test_deposit_zero_raises(account):
    with pytest.raises(ValueError):
        account.deposit(0.0)

def test_deposit_negative_raises(account):
    with pytest.raises(ValueError):
        account.deposit(-10.0)

def test_overdraft_raises(account):
    with pytest.raises(ValueError):
        account.withdraw(200.0)

def test_str(account):
    assert str(account) == "Account(Alice): €100.00"
```

## 2.3 — Run and verify

```bash
pytest tests/test_bank_account.py -v
```
