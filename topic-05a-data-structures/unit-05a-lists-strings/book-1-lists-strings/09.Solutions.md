## Exercise 3 - Vowels

~~~python
word = input("Enter a word: ").lower()
vowels = "aeiou"
count = 0

for letter in word:
    if letter in vowels:
        count = count + 1

print("Number of vowels:", count)
~~~

## Exercise 4 - Shopping list menu

~~~python
shopping = []

while True:
    print("1. Add item")
    print("2. View items")
    print("3. Remove item")
    print("4. Exit")

    choice = input("Choose: ")

    if choice == "1":
        item = input("Item: ")
        shopping.append(item)
    elif choice == "2":
        for item in shopping:
            print(item)
    elif choice == "3":
        item = input("Item to remove: ")
        if item in shopping:
            shopping.remove(item)
        else:
            print("Item not found")
    elif choice == "4":
        break
    else:
        print("Invalid choice")
~~~
