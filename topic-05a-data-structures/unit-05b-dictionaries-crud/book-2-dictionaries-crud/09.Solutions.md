## Update movie rating

~~~python
def update_rating():
    title = input("Title to update: ")

    for movie in movies:
        if movie["title"].lower() == title.lower():
            movie["rating"] = int(input("New rating: "))
            print("Rating updated")
            return

    print("Movie not found")
~~~

## Delete movie

~~~python
def delete_movie():
    title = input("Title to delete: ")

    for movie in movies:
        if movie["title"].lower() == title.lower():
            movies.remove(movie)
            print("Movie deleted")
            return

    print("Movie not found")
~~~

## Full menu choices

~~~python
while True:
    show_menu()
    choice = input("Choose: ")

    if choice == "1":
        add_movie()
    elif choice == "2":
        view_movies()
    elif choice == "3":
        search_movie()
    elif choice == "4":
        update_rating()
    elif choice == "5":
        delete_movie()
    elif choice == "6":
        break
    else:
        print("Invalid choice")
~~~
