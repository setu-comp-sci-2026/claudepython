## Challenge 1

~~~python
test1 = 72
test2 = 85
test3 = 91

print("Test 1:", test1)
print("Test 2:", test2)
print("Test 3:", test3)

total = test1 + test2 + test3
average = total / 3

print("Total:", total)
print("Average:", average)
~~~

## Challenge 2

~~~python
test1 = 72
test2 = 85
test3 = 91
bonus = 5

print("Test 1 with bonus:", test1 + bonus)
print("Test 2 with bonus:", test2 + bonus)
print("Test 3 with bonus:", test3 + bonus)

new_total = test1 + test2 + test3 + bonus * 3
new_average = new_total / 3

print("New Total:", new_total)
print("New Average:", new_average)
~~~

## Challenge 3

~~~python
price_a = 12
price_b = 7
price_c = 20

print("Item A: €", price_a)
print("Item B: €", price_b)
print("Item C: €", price_c)

total_one_each = price_a + price_b + price_c
print("Total (one of each): €", total_one_each)

total_custom = price_a * 2 + price_b * 1 + price_c * 3
print("Total (2A + 1B + 3C): €", total_custom)
~~~
