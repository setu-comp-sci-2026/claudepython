# Writing Your First Function

## Step 1 — Define and Call

```python
def say_hello():
    """Print a greeting."""
    print("Hello from my first function!")

say_hello()
say_hello()
say_hello()
```

Run the code. The function runs three times with a single definition.

## Step 2 — Function with a Parameter

```python
def greet(name):
    """Greet a person by name."""
    print(f"Hello, {name}! Welcome to Python.")

greet("Alice")
greet("Bob")
greet(input("Your name: "))
```

## Step 3 — Function that Returns a Value

```python
def square(n):
    """Return n squared."""
    return n * n

result = square(5)
print(f"5 squared = {result}")
print(f"10 squared = {square(10)}")
print(f"Sum of squares: {square(3) + square(4)}")  # 25
```

## Step 4 — Building a Mini Calculator

```python
def add(a, b):
    return a + b

def subtract(a, b):
    return a - b

def multiply(a, b):
    return a * b

def divide(a, b):
    if b == 0:
        return None     # can't divide by zero
    return a / b

# Use the functions
x, y = 10, 4
print(f"{x} + {y} = {add(x, y)}")
print(f"{x} - {y} = {subtract(x, y)}")
print(f"{x} × {y} = {multiply(x, y)}")
print(f"{x} ÷ {y} = {divide(x, y)}")
```
