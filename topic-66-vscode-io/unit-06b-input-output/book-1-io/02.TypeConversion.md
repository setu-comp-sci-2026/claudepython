# Type Conversion and Validation

## Step 1 — Type Conversion Functions

```python
# int() — whole numbers only
age = int("25")         # 25
# int("25.5")           # ValueError!

# float() — decimal numbers
price = float("19.99")  # 19.99
gpa   = float("3.8")    # 3.8

# str() — to string
num_str = str(42)       # "42"
pi_str  = str(3.14)     # "3.14"

# bool() — to boolean (rarely needed)
# Remember: non-zero/non-empty = True
print(bool(0))          # False
print(bool(1))          # True
print(bool(""))         # False
print(bool("hello"))    # True
```

## Step 2 — Basic try/except

```python
try:
    number = int(input("Enter a number: "))
    print(f"You entered: {number}")
    print(f"Doubled: {number * 2}")
except ValueError:
    print("That's not a valid integer!")
```

## Step 3 — Validated Input Loop

```python
def get_integer(prompt, minimum=None, maximum=None):
    """Keep asking until a valid integer is entered."""
    while True:
        try:
            value = int(input(prompt))
            if minimum is not None and value < minimum:
                print(f"Please enter a value >= {minimum}")
                continue
            if maximum is not None and value > maximum:
                print(f"Please enter a value <= {maximum}")
                continue
            return value
        except ValueError:
            print("Please enter a whole number")

age   = get_integer("Your age: ", 0, 120)
score = get_integer("Test score (0-100): ", 0, 100)
print(f"\nAge: {age}, Score: {score}")
```

## Step 4 — Yes/No Input

```python
def get_yes_no(prompt):
    """Return True for Y, False for N."""
    while True:
        response = input(f"{prompt} (Y/N): ").strip().upper()
        if response == "Y":
            return True
        elif response == "N":
            return False
        print("Please enter Y or N")

if get_yes_no("Do you want to continue?"):
    print("Continuing...")
else:
    print("Stopping.")
```
