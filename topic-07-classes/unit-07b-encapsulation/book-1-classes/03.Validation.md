# Adding Validation to Setters

## Step 1 — The Problem Without Validation

```python
class Person:
    def __init__(self, name, age):
        self.name = name
        self.age  = age

p = Person("Alice", 25)
p.age = -5          # Python allows this — but it's invalid!
p.name = ""         # An empty name — also invalid!
print(p.age)        # -5  (bad data slipped through)
```

## Step 2 — Setter Methods

```python
class Person:
    def __init__(self, name, age):
        self._name = None
        self._age  = None
        self.set_name(name)
        self.set_age(age)

    def get_name(self):
        return self._name

    def set_name(self, name):
        if not isinstance(name, str) or not name.strip():
            raise ValueError("Name must be a non-empty string")
        self._name = name.strip().title()

    def get_age(self):
        return self._age

    def set_age(self, age):
        if not isinstance(age, int):
            raise ValueError(f"Age must be an integer, got {type(age).__name__}")
        if age < 0 or age > 150:
            raise ValueError(f"Age must be 0-150, got {age}")
        self._age = age

    def __str__(self):
        return f"Person(name='{self._name}', age={self._age})"

p = Person("  alice  ", 25)
print(p)          # Person(name='Alice', age=25)  — name was cleaned
p.set_age(26)
print(p.get_age())
```

## Step 3 — Validate in __init__ Using Setters

A clean pattern: store data in private attributes but always go through the setter, even in `__init__`:

```python
class Student:
    def __init__(self, name, student_id, email):
        self._name       = None
        self._student_id = None
        self._email      = None

        # Use setters — validation runs immediately
        self.set_name(name)
        self.set_student_id(student_id)
        self.set_email(email)

    def set_name(self, name):
        if not name or not name.strip():
            raise ValueError("Name cannot be empty")
        self._name = name.strip().title()

    def set_student_id(self, sid):
        if not sid.startswith("S") or len(sid) != 4:
            raise ValueError(f"Student ID must be 'S' followed by 3 digits, got '{sid}'")
        self._student_id = sid

    def set_email(self, email):
        if "@" not in email or "." not in email.split("@")[-1]:
            raise ValueError(f"Invalid email: {email}")
        self._email = email.lower().strip()

    def __str__(self):
        return f"Student({self._student_id}: {self._name}, {self._email})"
```

## Step 4 — Testing Validation

```python
# Valid student
try:
    s = Student("Alice Murphy", "S001", "alice@example.com")
    print(s)
except ValueError as e:
    print(f"Unexpected error: {e}")

# Test each validation rule
invalid_cases = [
    ("", "S002", "bob@example.com"),          # empty name
    ("Carol", "X003", "carol@example.com"),   # bad ID format
    ("Dave", "S004", "not-an-email"),          # bad email
]

for name, sid, email in invalid_cases:
    try:
        s = Student(name, sid, email)
        print(f"ERROR: should have rejected ({name}, {sid}, {email})")
    except ValueError as e:
        print(f"Correctly rejected: {e}")
```
