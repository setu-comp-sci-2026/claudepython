# Creating a Student Class

## Step 1 — Define the Class

```python
class Student:
    """Represents a student in the system."""

    def __init__(self, name, student_id):
        """Initialise student with name and ID."""
        self.name       = name
        self.student_id = student_id
        self.scores     = []

    def __str__(self):
        return f"Student({self.student_id}: {self.name})"
```

## Step 2 — Create Objects

```python
alice = Student("Alice Murphy", "S001")
bob   = Student("Bob Kelly", "S002")

print(alice)        # Student(S001: Alice Murphy)
print(bob)          # Student(S002: Bob Kelly)
print(type(alice))  # <class '__main__.Student'>
```

## Step 3 — Add Methods

```python
class Student:
    def __init__(self, name, student_id):
        self.name       = name
        self.student_id = student_id
        self.scores     = []

    def add_score(self, score):
        """Add a score to the student's record."""
        self.scores.append(score)

    def average(self):
        """Return average score, or 0 if no scores recorded."""
        if not self.scores:
            return 0
        return sum(self.scores) / len(self.scores)

    def is_passing(self, threshold=60):
        """Return True if average meets the threshold."""
        return self.average() >= threshold

    def highest_score(self):
        """Return the highest score, or None if no scores."""
        return max(self.scores) if self.scores else None

    def __str__(self):
        avg = self.average()
        status = "Pass" if self.is_passing() else "Fail"
        return f"[{self.student_id}] {self.name}: avg={avg:.1f} ({status})"
```

## Step 4 — Use the Full Student Class

```python
alice = Student("Alice Murphy", "S001")
alice.add_score(85)
alice.add_score(92)
alice.add_score(78)

print(alice)                           # shows average and pass/fail
print(f"Highest: {alice.highest_score()}")
print(f"Passing: {alice.is_passing()}")
print(f"Passing at 90: {alice.is_passing(90)}")

# Create multiple students and find the best
students = [
    Student("Alice", "S001"),
    Student("Bob", "S002"),
    Student("Carol", "S003"),
]
for s, scores in zip(students, [[85, 90, 78], [62, 58, 70], [95, 88, 92]]):
    for score in scores:
        s.add_score(score)

best = max(students, key=lambda s: s.average())
print(f"\nTop student: {best}")
```
