# Converting to @property

## Step 1 — From Setters/Getters to @property

The `@property` decorator lets you keep validation but use attribute-style access.

```python
class Person:
    def __init__(self, name, age):
        self._name = None
        self._age  = None
        self.name  = name    # calls the setter
        self.age   = age     # calls the setter

    @property
    def name(self):
        return self._name

    @name.setter
    def name(self, value):
        if not isinstance(value, str) or not value.strip():
            raise ValueError("Name must be a non-empty string")
        self._name = value.strip().title()

    @property
    def age(self):
        return self._age

    @age.setter
    def age(self, value):
        if not isinstance(value, int) or not 0 <= value <= 150:
            raise ValueError(f"Age must be an integer 0-150, got {value}")
        self._age = value

    def __str__(self):
        return f"Person(name='{self.name}', age={self.age})"
```

## Step 2 — Using Properties

```python
p = Person("  alice  ", 25)
print(p)             # Person(name='Alice', age=25)

# Property access — looks like attributes
print(p.name)        # Alice
print(p.age)         # 25

# Property setter — validation runs
p.age = 26           # OK
p.name = "Bob"       # OK

try:
    p.age = -5       # ValueError
except ValueError as e:
    print(f"Error: {e}")
```

## Step 3 — Read-Only Properties

Omit the setter to make a property read-only:

```python
class Circle:
    def __init__(self, radius):
        self._radius = None
        self.radius = radius      # setter validates

    @property
    def radius(self):
        return self._radius

    @radius.setter
    def radius(self, value):
        if value <= 0:
            raise ValueError(f"Radius must be positive, got {value}")
        self._radius = value

    @property
    def area(self):
        """Read-only computed property."""
        import math
        return math.pi * self._radius ** 2

    @property
    def circumference(self):
        import math
        return 2 * math.pi * self._radius

    def __str__(self):
        return f"Circle(r={self.radius:.2f}, area={self.area:.2f})"

c = Circle(5)
print(c)
print(f"Area: {c.area:.4f}")
c.radius = 10
print(c)

# c.area = 100   # AttributeError — no setter
```

## Step 4 — BankAccount with Full Properties

```python
class BankAccount:
    def __init__(self, account_no, name, balance=0):
        self._account_no = account_no
        self._name       = None
        self._balance    = None
        self.name        = name      # use setter
        self.balance     = balance   # use setter

    @property
    def account_no(self):
        return self._account_no   # read-only

    @property
    def name(self):
        return self._name

    @name.setter
    def name(self, value):
        if not value.strip():
            raise ValueError("Name cannot be empty")
        self._name = value.strip()

    @property
    def balance(self):
        return self._balance

    @balance.setter
    def balance(self, value):
        if value < 0:
            raise ValueError(f"Balance cannot be negative: {value}")
        self._balance = value

    def deposit(self, amount):
        if amount <= 0:
            raise ValueError("Deposit must be positive")
        self.balance += amount

    def withdraw(self, amount):
        if amount <= 0:
            raise ValueError("Withdrawal must be positive")
        if amount > self.balance:
            raise ValueError("Insufficient funds")
        self.balance -= amount

    def __str__(self):
        return f"BankAccount({self.account_no}, {self.name}, €{self.balance:.2f})"

acc = BankAccount("ACC001", "Alice", 1000)
print(acc)
acc.deposit(500)
acc.withdraw(200)
print(acc)
```
