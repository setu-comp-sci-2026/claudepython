# Step 4 — Linear Search

Create `searching.py`:

```python
def linear_search(lst: list, target) -> int:
    """Return index of target, or -1 if not found."""
    for i, item in enumerate(lst):
        if item == target:
            return i
    return -1

# Test
data = [15, 8, 3, 42, 7, 19, 55, 1]
print(linear_search(data, 42))    # 3
print(linear_search(data, 100))   # -1
```

## Search with a key function

```python
def linear_search_by(lst: list, target, key) -> int:
    """Search using a key function."""
    for i, item in enumerate(lst):
        if key(item) == target:
            return i
    return -1

# Find product by name
idx = linear_search_by(products,
                        "Apple",
                        key=lambda p: p.get_name())
print(products[idx] if idx != -1 else "Not found")
```
