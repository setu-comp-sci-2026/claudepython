# Step 5 — Binary Search

```python
def binary_search(lst: list, target) -> int:
    """Binary search on a SORTED list.
    Returns index of target or -1."""
    low, high = 0, len(lst) - 1
    while low <= high:
        mid = (low + high) // 2
        if lst[mid] == target:
            return mid
        elif lst[mid] < target:
            low = mid + 1
        else:
            high = mid - 1
    return -1

# Must sort first!
data = sorted([15, 8, 3, 42, 7, 19, 55, 1])
print(data)                        # [1, 3, 7, 8, 15, 19, 42, 55]
print(binary_search(data, 19))     # 5
print(binary_search(data, 100))    # -1
```
