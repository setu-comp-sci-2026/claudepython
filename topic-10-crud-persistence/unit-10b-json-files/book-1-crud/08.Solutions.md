def add_product_menu(shop: Shop):

## Exercise 2 — JSON with Category

```python
# In Product
def to_dict(self) -> dict:
    return {
        "name": self.__name,
        "price": self.__price,
        "product_id": self.__product_id,
        "category": self.__category,
    }

@classmethod
def from_dict(cls, d: dict) -> "Product":
    p = cls(d["name"], d["price"], d["product_id"])
    p.set_category(d.get("category", "uncategorised"))
    return p
```

def save_to_json(self, filename: str = "shop.json"):

def load_from_json(self, filename: str = "shop.json"):
    with open(filename, "r") as f:
        data = json.load(f)
    self.__name = data["shop_name"]
    self.__products = []
    skipped = 0
    for item in data["products"]:
        try:
            self.__products.append(Product.from_dict(item))
        except (KeyError, ValueError) as e:
            print(f"  Skipped malformed entry: {e}")
            skipped += 1
    print(f"Loaded {len(self.__products)} products ({skipped} skipped).")
