## 2.1 — Wrap the Create menu action

```python
def add_product_menu(shop: Shop):
    try:
        name  = input("Name: ").strip()
        price = float(input("Price: €"))
        pid   = int(input("Product ID: "))
        shop.add_product(Product(name, price, pid))
        print("Product added.")
    except ValueError as e:
        print(f"  Error: {e}")
    except TypeError as e:
        print(f"  Type error: {e}")
```

## 2.2 — Wrap the Update menu action

```python
def update_product_menu(shop: Shop):
    try:
        pid = int(input("Product ID: "))
        new_price_str = input("New price (blank to skip): ").strip()
        new_price = float(new_price_str) if new_price_str else None
        if shop.update_product(pid, new_price=new_price):
            print("Updated.")
        else:
            print(f"ID {pid} not found.")
    except ValueError as e:
        print(f"  Error: {e}")
```

## 2.3 — try/except/else/finally example

```python
def load_data_safely(shop: Shop, filename: str):
    try:
        shop.load_from_json(filename)
    except FileNotFoundError:
        print("No saved data — starting fresh.")
    except Exception as e:
        print(f"Unexpected error: {e}")
    else:
        print("Data loaded successfully.")
    finally:
        print("Load attempt complete.")
```
