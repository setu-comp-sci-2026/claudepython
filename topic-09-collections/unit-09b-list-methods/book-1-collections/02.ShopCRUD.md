# Step 2 — Adding CRUD to the Menu

## 2.1 — Add Product (Create)

```python
def add_product_menu(shop: Shop):
    print("\n-- Add Product --")
    try:
        name  = input("Name: ").strip()
        price = get_float_input("Price (€): ")
        pid   = get_int_input("Product ID: ")
        shop.add_product(Product(name, price, pid))
        print(f"  '{name}' added successfully.")
    except ValueError as e:
        print(f"  Error: {e}")
```

## 2.2 — Find Product (Read)

```python
def find_product_menu(shop: Shop):
    print("\n-- Find Product --")
    name = input("Search by name: ").strip()
    result = shop.find_by_name(name)
    if result:
        print(f"  Found: {result}")
    else:
        print(f"  No product named '{name}'.")
```

## 2.3 — Update Price (add to Shop class first)

In `shop.py`:
```python
def update_price(self, name: str, new_price: float) -> bool:
    p = self.find_by_name(name)
    if p is None:
        return False
    p.set_price(new_price)
    return True
```

In `driver.py`:
```python
def update_price_menu(shop: Shop):
    name = input("Product to update: ").strip()
    try:
        new_price = get_float_input("New price (€): ")
        if shop.update_price(name, new_price):
            print("  Price updated.")
        else:
            print(f"  '{name}' not found.")
    except ValueError as e:
        print(f"  Error: {e}")
```

## 2.4 — Remove Product (Delete)

```python
def remove_product_menu(shop: Shop):
    name = input("Product to remove: ").strip()
    p = shop.find_by_name(name)
    if p is None:
        print(f"  '{name}' not found.")
        return
    confirm = input(f"  Remove '{p.get_name()}'? (y/n): ").strip().lower()
    if confirm == "y":
        shop.remove_product(name)
        print("  Removed.")
    else:
        print("  Cancelled.")
```

## 2.5 — Update `main()` to wire everything up

```python
while True:
```
