# Step 1 — Building a Menu Loop

## 1.1 — Define `show_menu()`

In your `driver.py`:

```python
def show_menu():
    print("\n=== Shop v3.0 ===")
    print("1. Add product")
    print("2. List all products")
    print("3. Find product by name")
    print("4. Remove product")
    print("0. Exit")
```

## 1.2 — The main loop

```python
def main():
    shop = Shop("My Shop")
    # Add some starter data
    shop.add_product(Product("Apple",  1.99, 101))
    shop.add_product(Product("Bread",  2.50, 102))
    shop.add_product(Product("Milk",   1.20, 103))

    while True:
        show_menu()
        choice = input("Enter choice: ").strip()
        match choice:
            case "1": print("(Add not yet implemented)")
            case "2": shop.display_all()
            case "3": print("(Find not yet implemented)")
            case "4": print("(Remove not yet implemented)")
            case "0":
                print("Goodbye!")
                break
            case _:
                print("Invalid choice — try again.")

if __name__ == "__main__":
    main()
```

Run it and test the `List all` and `Exit` options.

## 1.3 — Safe integer input helper

```python
def get_int_input(prompt: str) -> int:
    while True:
        try:
            return int(input(prompt))
        except ValueError:
            print("Please enter a whole number.")

def get_float_input(prompt: str) -> float:
    while True:
        try:
            val = float(input(prompt))
            if val > 0:
                return val
            print("Must be greater than 0.")
        except ValueError:
            print("Please enter a number.")
```
