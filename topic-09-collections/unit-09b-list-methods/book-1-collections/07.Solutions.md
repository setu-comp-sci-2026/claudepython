## Exercise 1 — Price Range Filter

```python
# In shop.py
def get_in_price_range(self, min_price: float, max_price: float) -> list:
    return [p for p in self.__products
            if min_price <= p.get_price() <= max_price]

# In driver.py
def price_range_menu(shop: Shop):
    min_p = get_float_input("Min price: €")
    max_p = get_float_input("Max price: €")
    results = shop.get_in_price_range(min_p, max_p)
    if results:
        for p in results:
            print(f"  {p}")
    else:
        print("  No products in that range.")
```

## Exercise 2 — Search by Initial Letter

```python
def search_by_letter(shop: Shop):
    letter = input("Enter first letter: ").strip().lower()
    if not letter:
        return
    matches = [p for p in shop.get_all_products()
               if p.get_name().lower().startswith(letter)]
    print(f"\nProducts starting with '{letter}':")
    for p in matches:
        print(f"  {p}")
    if not matches:
        print("  None found.")
```

## Exercise 3 — Total by Category

```python
def total_by_category(shop: Shop, category: str) -> float:
    return sum(p.get_price()
               for p in shop.get_all_products()
               if p.get_category() == category.lower())
```
