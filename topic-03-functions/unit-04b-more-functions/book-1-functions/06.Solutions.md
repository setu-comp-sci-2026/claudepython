## Exercise 1 — Temperature Converter

```python
def celsius_to_fahrenheit(c):
    return (c * 9 / 5) + 32

def fahrenheit_to_celsius(f):
    return (f - 32) * 5 / 9

while True:
    print("\n1. Celsius to Fahrenheit")
    print("2. Fahrenheit to Celsius")
    print("3. Quit")
    choice = input("Choice: ")

    if choice == "1":
        c = float(input("Celsius: "))
        print(f"{c:.1f}C = {celsius_to_fahrenheit(c):.1f}F")
    elif choice == "2":
        f = float(input("Fahrenheit: "))
        print(f"{f:.1f}F = {fahrenheit_to_celsius(f):.1f}C")
    elif choice == "3":
        break
    else:
        print("Invalid choice")
```

## Exercise 2 — Palindrome Checker

```python
def is_palindrome(word):
    cleaned = "".join(c.lower() for c in word if c.isalnum())
    return cleaned == cleaned[::-1]

tests = ["racecar", "hello", "Madam", "Never odd or even"]
for word in tests:
    result = "palindrome" if is_palindrome(word) else "not a palindrome"
    print(f"'{word}' is {result}")
```

## Exercise 3 — Word Counter

```python
def word_stats(text):
    words = text.lower().split()
    if not words:
        return {"words": 0, "unique": 0, "longest": "", "avg_length": 0}
    return {
        "words": len(words),
        "unique": len(set(words)),
        "longest": max(words, key=len),
        "avg_length": round(sum(len(w) for w in words) / len(words), 2),
    }

stats = word_stats("the quick brown fox jumps over the lazy dog")
for key, value in stats.items():
    print(f"{key}: {value}")
```

## Exercise 4 — Number Formatter

```python
def format_number(n):
    if 0 <= n < 1:
        return f"{n:.2%}"
    if isinstance(n, int) or n == int(n):
        return f"{int(n):,}"
    return f"{n:,.2f}"

print(format_number(1234567))   # 1,234,567
print(format_number(9876.543))  # 9,876.54
print(format_number(0.7523))    # 75.23%
```

## Challenge 1 — Recursive Sum

```python
def recursive_sum(numbers):
    if not numbers:
        return 0
    return numbers[0] + recursive_sum(numbers[1:])

def recursive_sum_nested(numbers):
    if not numbers:
        return 0
    first = numbers[0]
    rest = recursive_sum_nested(numbers[1:])
    if isinstance(first, list):
        return recursive_sum_nested(first) + rest
    return first + rest

print(recursive_sum([1, 2, 3, 4, 5]))          # 15
print(recursive_sum_nested([1, [2, 3], [4, [5]]]))  # 15
```

## Challenge 2 — Caesar Cipher

```python
def encrypt(text, shift):
    result = ""
    for char in text:
        if char.isalpha():
            base = ord('a') if char.islower() else ord('A')
            shifted = (ord(char) - base + shift) % 26 + base
            result += chr(shifted)
        else:
            result += char
    return result

def decrypt(text, shift):
    return encrypt(text, -shift)

message = "Hello, World!"
encrypted = encrypt(message, 13)
print(f"Encrypted: {encrypted}")
print(f"Decrypted: {decrypt(encrypted, 13)}")
```

## Challenge 3 — Word Frequency Counter

```python
def word_frequency(text):
    words = text.lower().split()
    freq = {}
    for word in words:
        freq[word] = freq.get(word, 0) + 1
    return sorted(freq.items(), key=lambda x: (-x[1], x[0]))

text = "the quick brown fox jumps over the lazy dog the fox"
for word, count in word_frequency(text):
    print(f"{word:<12} {count}")
```
