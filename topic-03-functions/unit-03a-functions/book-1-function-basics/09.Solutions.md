## Exercise 2 (BMI)

~~~python
def bmi(weight_kg, height_m):
    """Calculate and return the Body Mass Index."""
    return weight_kg / (height_m ** 2)

def bmi_category(bmi_value):
    """Return the BMI category string for a given BMI value."""
    if bmi_value < 18.5: return "Underweight"
    if bmi_value < 25.0: return "Normal weight"
    if bmi_value < 30.0: return "Overweight"
    return "Obese"

weight = float(input("Weight (kg): "))
height = float(input("Height (m):  "))
value  = bmi(weight, height)
print(f"BMI: {value:.1f} — {bmi_category(value)}")
~~~


## Exercise 3 (void vs return)

~~~python
# double_print() returns None because it uses print() instead of return.
# double_return() returns the actual value 10, which can be used later.

def double_print(n):
    print(n * 2)

def double_return(n):
    return n * 2

a = double_print(5)   # prints 10, a is None
b = double_return(5)  # b is 10

print(a)          # None
print(b)          # 10
print(type(a))    # <class 'NoneType'>
print(type(b))    # <class 'int'>
~~~


## Exercise 4 (show_all)

~~~python
def add(a, b):
    """Return the sum of a and b."""
    return a + b

def subtract(a, b):
    """Return a minus b."""
    return a - b

def multiply(a, b):
    """Return the product of a and b."""
    return a * b

def divide(a, b):
    """Return a divided by b, or None if b is zero."""
    if b == 0:
        print("Warning: cannot divide by zero")
        return None
    return a / b

def show_all(a, b):
    """Print the result of all four operations on a and b."""
    print(f"{a} + {b}  = {add(a, b)}")
    print(f"{a} - {b}  = {subtract(a, b)}")
    print(f"{a} × {b}  = {multiply(a, b)}")
    result = divide(a, b)
    if result is not None:
        print(f"{a} ÷ {b}  = {result}")

show_all(10, 4)
~~~


## Challenge 1 (number guessing game)

~~~python
import random

def get_secret_number():
    """Generate and return a random integer between 1 and 100."""
    return random.randint(1, 100)

def get_guess():
    """Prompt the user for a guess and return it as an integer."""
    return int(input("Guess the number (1-100): "))

def check_guess(secret, guess):
    """Check the guess and return True if correct, False otherwise."""
    if guess < secret:
        print("Too low")
        return False
    if guess > secret:
        print("Too high")
        return False
    print("Correct!")
    return True

secret   = get_secret_number()
attempts = 0

while True:
    guess = get_guess()
    attempts += 1
    if check_guess(secret, guess):
        print(f"You got it in {attempts} attempt{'s' if attempts > 1 else ''}.")
        break
~~~


## Challenge 2 (class report)

~~~python
def get_scores():
    """Collect names and scores for a class."""
    n = int(input("How many students? "))
    names, scores = [], []
    for _ in range(n):
        names.append(input("  Name:  "))
        scores.append(float(input("  Score: ")))
    return names, scores

def class_average(scores):
    """Return the average of a list of scores."""
    return sum(scores) / len(scores)

def highest_scorer(names, scores):
    """Return the name of the student with the highest score."""
    return names[scores.index(max(scores))]

def grade(score):
    if score >= 90: return "A"
    if score >= 80: return "B"
    if score >= 70: return "C"
    if score >= 60: return "D"
    return "F"

def print_report(names, scores):
    """Print a formatted class report."""
    print("=" * 30)
    print(f"{'CLASS REPORT':^30}")
    print("=" * 30)
    for name, score in zip(names, scores):
        print(f"{name:<12} {score:>6.1f}   {grade(score)}")
    print("-" * 30)
    print(f"Average:     {class_average(scores):.1f}")
    print(f"Top student: {highest_scorer(names, scores)}")
    print("=" * 30)

names, scores = get_scores()
print_report(names, scores)
~~~

~~~python
import math

def circle_area(radius):
    """Return the area of a circle."""
    return math.pi * radius ** 2

def circle_circumference(radius):
    """Return the circumference of a circle."""
    return 2 * math.pi * radius

def rectangle_area(w, h):
    """Return the area of a rectangle."""
    return w * h

def rectangle_perimeter(w, h):
    """Return the perimeter of a rectangle."""
    return 2 * (w + h)

def triangle_area(base, height):
    """Return the area of a triangle."""
    return 0.5 * base * height
~~~
