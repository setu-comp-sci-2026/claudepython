## Exercise 2 (circle_stats)

~~~python
import math

def circle_stats(radius):
    """Return the area and circumference of a circle."""
    area = math.pi * radius ** 2
    circumference = 2 * math.pi * radius
    return area, circumference

for r in [5, 10, 2.5]:
    area, circ = circle_stats(r)
    print(f"Radius: {r}")
    print(f"Area:          {area:.2f}")
    print(f"Circumference: {circ:.2f}")
    print()
~~~


## Exercise 3 (ternary functions)

~~~python
def is_positive(n):
    """Return 'positive' or 'non-positive'."""
    return "positive" if n > 0 else "non-positive"

def larger(a, b):
    """Return the larger of a and b."""
    return a if a > b else b

def clamp(value, low, high):
    """Clamp value to the range [low, high]."""
    return low if value < low else (high if value > high else value)

print(is_positive(5))       # positive
print(is_positive(-3))      # non-positive
print(larger(7, 4))         # 7
print(larger(2, 9))         # 9
print(clamp(15, 0, 10))     # 10
print(clamp(-3, 0, 10))     # 0
print(clamp(5, 0, 10))      # 5
~~~


## Exercise 4 (shipping_cost)

~~~python
def shipping_cost(weight_kg):
    """Return shipping cost based on weight in kg."""
    if weight_kg <= 1:  return 3.00
    if weight_kg <= 5:  return 5.50
    if weight_kg <= 20: return 9.00
    return 15.00

def print_quote(weight_kg):
    """Print a formatted shipping quote."""
    cost = shipping_cost(weight_kg)
    print(f"Weight: {weight_kg} kg  →  Shipping: €{cost:.2f}")

for w in [0.5, 2, 10, 25]:
    print_quote(w)
~~~


## Challenge 1 (invoice)

~~~python
def line_total(qty, unit_price, discount=0.0):
    """Return the total for one line item after discount."""
    return qty * unit_price * (1 - discount)

def apply_vat(amount, rate=0.23):
    """Return amount with VAT applied."""
    return amount * (1 + rate)

def format_money(amount):
    """Return amount formatted as a euro string."""
    return f"€{amount:.2f}"

def print_invoice(items):
    """Print a formatted invoice for a list of (name, qty, price) tuples."""
    width = 30
    print("=" * width)
    print(f"{'INVOICE':^{width}}")
    print("=" * width)

    subtotal = 0
    for name, qty, price in items:
        total = line_total(qty, price)
        subtotal += total
        print(f"{name:<12} {qty} x {format_money(price):>6}  = {format_money(total):>6}")

    vat_amount = subtotal * 0.23
    grand_total = apply_vat(subtotal)

    print("-" * width)
    print(f"{'Subtotal:':20} {format_money(subtotal):>9}")
    print(f"{'VAT (23%):':20} {format_money(vat_amount):>9}")
    print("=" * width)
    print(f"{'Total:':20} {format_money(grand_total):>9}")
    print("=" * width)

items = [("Coffee", 3, 3.50), ("Sandwich", 2, 5.00), ("Water", 4, 1.20)]
print_invoice(items)
~~~


## Challenge 2 (statistics library)

~~~python
import math

def mean(numbers):
    """Return the arithmetic mean."""
    return sum(numbers) / len(numbers)

def median(numbers):
    """Return the median value."""
    sorted_nums = sorted(numbers)
    n = len(sorted_nums)
    mid = n // 2
    if n % 2 == 0:
        return (sorted_nums[mid - 1] + sorted_nums[mid]) / 2
    return sorted_nums[mid]

def mode(numbers):
    """Return the most frequently occurring value."""
    return max(set(numbers), key=numbers.count)

def std_dev(numbers):
    """Return the population standard deviation."""
    avg = mean(numbers)
    variance = mean([(x - avg) ** 2 for x in numbers])
    return math.sqrt(variance)

def full_report(numbers):
    """Print a statistical summary of a list of numbers."""
    print(f"Count:  {len(numbers)}")
    print(f"Mean:   {mean(numbers):.2f}")
    print(f"Median: {median(numbers):.2f}")
    print(f"Mode:   {mode(numbers):.2f}")
    print(f"Std Dev:{std_dev(numbers):.2f}")

full_report([4, 7, 2, 9, 4, 1, 7, 4, 8, 3])
~~~


## Challenge 3 (refactored score program)

~~~python
def get_score(prompt="Enter score (or 'done'): "):
    """Prompt for a score string and return it."""
    return input(prompt)

def is_valid_score(score):
    """Return True if score is in range 0-100."""
    return 0 <= score <= 100

def collect_scores():
    """Collect valid scores from the user and return as a list."""
    scores = []
    while True:
        raw = get_score()
        if raw.lower() == "done":
            break
        score = float(raw)
        if not is_valid_score(score):
            print("Invalid score — must be 0-100")
            continue
        scores.append(score)
    return scores

def calculate_grade(average):
    """Return a letter grade for a numeric average using chained returns."""
    if average >= 90: return "A"
    if average >= 80: return "B"
    if average >= 70: return "C"
    if average >= 60: return "D"
    return "F"

def summarise(scores):
    """Return a dict of summary statistics for a list of scores."""
    avg = sum(scores) / len(scores)
    return {
        "count":   len(scores),
        "average": avg,
        "highest": max(scores),
        "lowest":  min(scores),
        "grade":   calculate_grade(avg),
    }

def display_summary(stats):
    """Print a formatted summary from a stats dict."""
    print(f"Count:   {stats['count']}")
    print(f"Average: {stats['average']:.1f}  Grade: {stats['grade']}")
    print(f"Highest: {stats['highest']}  Lowest: {stats['lowest']}")

# Main program
scores = collect_scores()
if not scores:
    print("No scores entered.")
else:
    display_summary(summarise(scores))
~~~
