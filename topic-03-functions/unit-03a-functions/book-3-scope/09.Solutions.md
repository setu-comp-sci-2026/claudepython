## Exercise 1 (local vs global)

~~~python
# Predictions:
# double()  → Inside double: 20   (local x shadows global)
# triple()  → Inside triple: 10   (reads global x)
# Global    → Global: 10          (global x unchanged)
#
# double() does not change the global x because assigning x = 20
# inside the function creates a NEW local variable called x.
# It does not write back to the global x.

x = 10

def double():
    x = 20
    print(f"Inside double: {x}")

def triple():
    print(f"Inside triple: {x}")

double()
triple()
print(f"Global: {x}")
~~~

## Exercise 3 (pure vs impure)

~~~python
import random

def add(a, b):
    # PURE — same inputs always give same output, no side effects
    return a + b

def get_name():
    # IMPURE — depends on user input (external state)
    return input("Name: ")

# Pure equivalent: pass the name as a parameter
def process_name(name):
    # PURE — same input always gives same output
    return name.strip()

def to_uppercase(text):
    # PURE — same input always gives same output, no side effects
    return text.upper()

def random_greeting():
    # IMPURE — different result on each call (uses random)
    options = ["Hi", "Hello", "Hey"]
    return random.choice(options)

# Pure equivalent: accept the choice as a parameter
def format_greeting(greeting, name):
    # PURE — deterministic, no side effects
    return f"{greeting}, {name}!"

def print_total(total):
    # IMPURE — side effect (printing to screen)
    print(f"Total: {total}")

# Pure equivalent: return a formatted string instead
def format_total(total):
    # PURE — returns a value, no side effects
    return f"Total: {total}"
~~~


## Exercise 4 (constants and intermediate variables)

~~~python
GRAVITY        = 9.81
SPEED_OF_LIGHT = 299_792_458
AIR_DENSITY    = 1.225

def free_fall_velocity(time_s):
    """Return velocity after free-falling for time_s seconds."""
    velocity = GRAVITY * time_s
    return velocity

def kinetic_energy(mass_kg, velocity_ms):
    """Return kinetic energy in joules."""
    velocity_squared = velocity_ms ** 2
    energy           = 0.5 * mass_kg * velocity_squared
    return energy

def drag_force(area_m2, velocity_ms, drag_coefficient=0.47):
    """Return aerodynamic drag force in newtons."""
    velocity_squared = velocity_ms ** 2
    dynamic_pressure = 0.5 * AIR_DENSITY * velocity_squared
    force            = dynamic_pressure * area_m2 * drag_coefficient
    return force

v = free_fall_velocity(5)
print(f"Velocity after 5s free fall:  {v:.2f} m/s")
print(f"Kinetic energy (70kg, {v:.1f}m/s): {kinetic_energy(70, v):.1f} J")
print(f"Drag on 0.5m² at {v:.1f}m/s:      {drag_force(0.5, v):.2f} N")
~~~


## Challenge 1 (refactored score tracker)

~~~python
def get_score(prompt="Enter score: "):
    """Prompt for a score and return it as a float."""
    return float(input(prompt))

def compute_average(total, count):
    """Return the average of total over count entries."""
    return total / count

def display_average(average):
    """Print the running average."""
    print(f"Running average: {average:.1f}")

def collect_scores(n):
    """Collect n scores and return the final total and count."""
    total = 0
    count = 0
    for _ in range(n):
        score  = get_score()
        total  = total + score
        count  = count + 1
        avg    = compute_average(total, count)
        display_average(avg)
    return total, count

collect_scores(3)
~~~


## Challenge 2 (mortgage module)

~~~python
MONTHS_PER_YEAR = 12
STAMP_DUTY_RATE = 0.01

def monthly_rate(annual_rate):
    """Return the monthly interest rate."""
    return annual_rate / MONTHS_PER_YEAR

def n_payments(years):
    """Return the total number of monthly payments."""
    return years * MONTHS_PER_YEAR

def monthly_payment(principal, annual_rate, years):
    """Return the monthly repayment amount."""
    r      = monthly_rate(annual_rate)
    n      = n_payments(years)
    factor = (1 + r) ** n
    return principal * r * factor / (factor - 1)

def stamp_duty(price):
    """Return the stamp duty on a property."""
    return price * STAMP_DUTY_RATE

def total_cost(principal, annual_rate, years):
    """Return total amount repaid over the full term."""
    payment    = monthly_payment(principal, annual_rate, years)
    n_months   = n_payments(years)
    return payment * n_months

def print_mortgage_summary(price, deposit, annual_rate, years):
    """Print a formatted mortgage summary."""
    principal = price - deposit
    payment   = monthly_payment(principal, annual_rate, years)
    total     = total_cost(principal, annual_rate, years)
    duty      = stamp_duty(price)
    width     = 30
    print("=" * width)
    print(f"{'MORTGAGE SUMMARY':^{width}}")
    print("=" * width)
    print(f"Property price:  €{price:>10,.0f}")
    print(f"Deposit:         €{deposit:>10,.0f}")
    print(f"Loan amount:     €{principal:>10,.0f}")
    print(f"Annual rate:     {annual_rate * 100:>10.2f}%")
    print(f"Term:            {years:>9} years")
    print("-" * width)
    print(f"Monthly payment: €{payment:>10,.2f}")
    print(f"Total repaid:    €{total:>10,.2f}")
    print(f"Stamp duty:      €{duty:>10,.2f}")
    print("=" * width)

print_mortgage_summary(350_000, 35_000, 0.04, 30)
~~~


## Challenge 3 (unit converter)

~~~python
KG_PER_LB   = 0.453592
M_PER_FOOT  = 0.3048
KM_PER_MILE = 1.60934
C_OFFSET    = 32
C_SCALE     = 5 / 9

def lbs_to_kg(lbs):
    return lbs * KG_PER_LB

def kg_to_lbs(kg):
    return kg / KG_PER_LB

def feet_to_metres(feet):
    return feet * M_PER_FOOT

def metres_to_feet(m):
    return m / M_PER_FOOT

def miles_to_km(miles):
    return miles * KM_PER_MILE

def km_to_miles(km):
    return km / KM_PER_MILE

def fahrenheit_to_celsius(f):
    adjusted = f - C_OFFSET
    return adjusted * C_SCALE

def celsius_to_fahrenheit(c):
    scaled = c / C_SCALE
    return scaled + C_OFFSET

CONVERTERS = {
    "lbs":        (lbs_to_kg,           "kg"),
    "feet":       (feet_to_metres,       "m"),
    "miles":      (miles_to_km,          "km"),
    "fahrenheit": (fahrenheit_to_celsius, "°C"),
}

def conversion_table(values, from_unit):
    """Print a two-column conversion table."""
    convert, to_unit = CONVERTERS[from_unit]
    print(f"{from_unit:<12} {to_unit}")
    print("-" * 12 + "  " + "-" * 12)
    for v in values:
        print(f"{v:<12} {convert(v):.2f}")

conversion_table([100, 150, 200], "lbs")
print()
conversion_table([32, 72, 100, 212], "fahrenheit")
~~~
