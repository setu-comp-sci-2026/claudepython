## Exercise 1 (normalise)

~~~python
def normalise(text):
    """Strip whitespace, title-case, and collapse double spaces."""
    # split() on no args splits on any whitespace and discards empties
    # join() with a single space collapses all runs of whitespace
    return " ".join(text.split()).title()

print(normalise("  alice   smith  "))   # Alice Smith
print(normalise("JOHN  DOE"))           # John Doe
print(normalise("  mary   "))           # Mary
~~~

def clean_phone(number):
    """Remove spaces, dashes, and brackets from a phone number string."""
    return number.replace(" ", "").replace("-", "").replace("(", "").replace(")", "")

print(clean_phone("(01) 234-5678"))     # 0123456789
print(clean_phone("+353 86 123 4567"))  # +353861234567
print(clean_phone("086-555-1234"))      # 0865551234

## Exercise 3 (initials and word_count)

~~~python
def initials(full_name):
    """Return dot-separated uppercase initials from a full name (2 or 3 words)."""
    parts = full_name.strip().split()
    result = parts[0][0].upper()
    if len(parts) > 1:
        result = result + "." + parts[1][0].upper()
    if len(parts) > 2:
        result = result + "." + parts[2][0].upper()
    return result

def word_count(text):
    """Return a dict with total words, unique words, and the longest word."""
    words = text.lower().split()
    return {
        "words":   len(words),
        "unique":  len(set(words)),
        "longest": max(words, key=len),
    }

print(initials("alice mary smith"))     # A.M.S
print(initials("John Doe"))             # J.D
print(initials("  bob   jones  "))      # B.J

stats = word_count("the quick brown fox jumps over the lazy dog")
print(stats)
~~~


## Exercise 4 (format_item and print_receipt)

~~~python
def format_item(name, qty, unit_price):
    """Return a formatted receipt line for one item."""
    total = qty * unit_price
    return f"{name:<15} {qty:>2}   €{total:>6.2f}"

def print_receipt(title, name1, qty1, price1, name2, qty2, price2, name3, qty3, price3):
    """Print a formatted three-item receipt."""
    width = 40
    total = qty1 * price1 + qty2 * price2 + qty3 * price3
    print("=" * width)
    print(f"{title:^{width}}")
    print("=" * width)
    print(format_item(name1, qty1, price1))
    print(format_item(name2, qty2, price2))
    print(format_item(name3, qty3, price3))
    print("-" * width)
    print(f"{'Total':>27}   €{total:>6.2f}")
    print("=" * width)

print_receipt("CAFÉ RECEIPT",
              "Coffee", 2, 3.50,
              "Cake",   1, 4.75,
              "Tea",    3, 2.00)
~~~


## Challenge 1 (validate_email)

~~~python
def validate_email(email):
    """Clean and validate an email address. Returns (email, message) tuple."""
    cleaned = email.strip().lower()
    if "@" not in cleaned:
        return None, "Missing @ symbol"
    username, _, domain = cleaned.partition("@")
    if "." not in domain:
        return None, "Missing domain"
    if len(username) < 2:
        return None, "Username too short"
    return cleaned, "Valid"

print(validate_email("  Alice@Example.COM  "))
print(validate_email("notanemail"))
print(validate_email("missing@dot"))
print(validate_email("a@b.com"))
print(validate_email("  VALID.USER@domain.ie  "))
~~~


## Challenge 2 (URL parser)

~~~python
def get_scheme(url):
    """Return the scheme from a URL (e.g. 'https')."""
    return url.split("://")[0]

def get_domain(url):
    """Return the domain from a URL (e.g. 'www.example.com')."""
    after_scheme = url.split("://")[1]          # 'www.example.com/path?q=1'
    return after_scheme.split("/")[0]           # 'www.example.com'

def get_path(url):
    """Return the path from a URL (e.g. '/search/results')."""
    after_scheme = url.split("://")[1]          # 'www.example.com/path?q=1'
    after_domain = after_scheme[len(get_domain(url)):]  # '/path?q=1'
    path, _, _   = after_domain.partition("?") # '/path'
    return path

def get_query(url):
    """Return the query string from a URL, or '' if none."""
    _, sep, query = url.partition("?")
    return query

url = "https://www.example.com/search/results?q=python&page=2"
print(get_scheme(url))   # https
print(get_domain(url))   # www.example.com
print(get_path(url))     # /search/results
print(get_query(url))    # q=python&page=2

url2 = "http://docs.python.org/3/library/string.html"
print(get_scheme(url2))  # http
print(get_domain(url2))  # docs.python.org
print(get_path(url2))    # /3/library/string.html
print(get_query(url2))   # (empty string)
~~~


## Challenge 3 (Caesar cipher)

~~~python
import string

def caesar_cipher(text, shift):
    """Encrypt text using Caesar cipher with the given shift."""
    shift     = shift % 26
    lower     = string.ascii_lowercase
    upper     = string.ascii_uppercase
    rot_lower = lower[shift:] + lower[:shift]
    rot_upper = upper[shift:] + upper[:shift]
    table     = str.maketrans(lower + upper, rot_lower + rot_upper)
    return text.translate(table)

message   = "Hello, World!"
encrypted = caesar_cipher(message, 13)
print(encrypted)   # Uryyb, Jbeyq!

decrypted = caesar_cipher(encrypted, 13)
print(decrypted)   # Hello, World!
~~~
