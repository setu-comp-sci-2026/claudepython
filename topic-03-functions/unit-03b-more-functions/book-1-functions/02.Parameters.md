# Parameters and Return Values

## Step 1 — Default Parameters

```python
def power(base, exponent=2):
    """Raise base to a given power (default: squared)."""
    return base ** exponent

print(power(3))         # 9  (uses default exponent=2)
print(power(3, 3))      # 27
print(power(2, 10))     # 1024
```

## Step 2 — Keyword Arguments

```python
def describe_person(name, age, city="Unknown"):
    print(f"{name} is {age} years old and lives in {city}.")

describe_person("Alice", 25, "Dublin")
describe_person("Bob", 30)                   # city defaults to "Unknown"
describe_person(age=22, name="Carol", city="Cork")
```

## Step 3 — Multiple Return Values

```python
def statistics(numbers):
    """Return count, total, minimum, maximum, and average."""
    count = len(numbers)
    total = sum(numbers)
    minimum = min(numbers)
    maximum = max(numbers)
    average = total / count
    return count, total, minimum, maximum, average

scores = [72, 85, 91, 64, 78, 88, 95, 70]
n, tot, lo, hi, avg = statistics(scores)
print(f"Count:   {n}")
print(f"Total:   {tot}")
print(f"Min:     {lo}")
print(f"Max:     {hi}")
print(f"Average: {avg:.1f}")
```

## Step 4 — Guard Clause Pattern

```python
def safe_divide(numerator, denominator):
    """Divide two numbers, returning None if denominator is zero."""
    if denominator == 0:
        print("Error: cannot divide by zero")
        return None     # early return (guard clause)
    return numerator / denominator

result = safe_divide(10, 2)
if result is not None:
    print(f"Result: {result}")
```
