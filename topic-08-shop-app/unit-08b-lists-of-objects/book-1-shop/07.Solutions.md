## Exercise 1 — Extend Product (quantity)

```python
class Product:
    def __init__(self, name: str, price: float,
                 product_id: int, quantity: int = 0):
        self.__product_id = product_id
        self.__name = ""
        self.__price = 0.0
        self.__quantity = 0
        self.set_name(name)
        self.set_price(price)
        self.set_quantity(quantity)

    def get_quantity(self) -> int:
        return self.__quantity

    def set_quantity(self, quantity: int):
        if quantity < 0:
            raise ValueError("Quantity cannot be negative")
        self.__quantity = quantity

    def __str__(self) -> str:
        return (f"Product [{self.__product_id}]: "
                f"{self.__name} — €{self.__price:.2f} "
                f"(qty: {self.__quantity})")
```

## Exercise 2 — find_by_id

```python
def find_by_id(self, product_id: int):
    for p in self.__products:
        if p.get_product_id() == product_id:
            return p
    return None
```

Driver test:
```python
p = my_shop.find_by_id(102)
print(p if p else "Not found")
```
