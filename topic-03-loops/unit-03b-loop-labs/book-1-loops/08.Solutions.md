## Exercise 2 — Sum of Digits

```python
number = int(input("Enter a positive integer: "))
digit_sum = 0
for char in str(number):
    digit_sum += int(char)
print(f"Sum of digits of {number}: {digit_sum}")
```

## Exercise 3 — Number Pyramid

```python
n = int(input("Enter number of rows: "))
for row in range(1, n + 1):
    spaces = " " * (n - row)
    digits = "".join(str(d) for d in range(1, row + 1))
    print(spaces + digits)
```

## Exercise 4 — Word Statistics

```python
sentence = input("Enter a sentence: ")
words = sentence.split()
word_count = len(words)
char_count = sum(len(w) for w in words)
vowel_count = sum(1 for c in sentence.lower() if c in "aeiou")
longest = max(words, key=len)

print(f"Words:        {word_count}")
print(f"Characters:   {char_count}")
print(f"Vowels:       {vowel_count}")
print(f"Longest word: {longest}")
```

## Challenge 1 — Prime Number Checker

```python
def is_prime(n):
    if n < 2:
        return False
    for divisor in range(2, int(n**0.5) + 1):
        if n % divisor == 0:
            return False
    return True

primes = [n for n in range(1, 101) if is_prime(n)]
print("Primes from 1 to 100:")
print(" ".join(str(p) for p in primes))
print(f"Count: {len(primes)}")
```

## Challenge 2 — Fibonacci Sequence

```python
count = int(input("How many Fibonacci numbers? "))

fibs = []
a, b = 0, 1
for _ in range(count):
    fibs.append(a)
    a, b = b, a + b

print("Fibonacci sequence:")
print(", ".join(str(f) for f in fibs))

print("\nGolden ratio approach:")
for i in range(2, min(count, 20)):
    if fibs[i - 1] != 0:
        ratio = fibs[i] / fibs[i - 1]
        print(f"  {fibs[i]} / {fibs[i-1]} = {ratio:.6f}")
```

## Challenge 3 — Number Guessing Game

```python
import random

while True:
    secret = random.randint(1, 100)
    max_guesses = 7
    print(f"\nI've picked a number between 1 and 100. You have {max_guesses} guesses.")

    for attempt in range(1, max_guesses + 1):
        guess = int(input(f"Guess {attempt}/{max_guesses}: "))
        if guess < secret:
            print("Too low!")
        elif guess > secret:
            print("Too high!")
        else:
            print(f"Correct! You guessed it in {attempt} attempt(s)!")
            break
    else:
        print(f"Out of guesses! The number was {secret}.")

    again = input("Play again? (Y/N): ").upper()
    if again != "Y":
        print("Thanks for playing!")
        break
```
